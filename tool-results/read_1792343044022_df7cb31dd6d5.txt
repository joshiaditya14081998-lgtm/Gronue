   173→
   174→function Reveal({
   175→  children,
   176→  delay = 0,
   177→  className = '',
   178→}: {
   179→  children: React.ReactNode
   180→  delay?: number
   181→  className?: string
   182→}) {
   183→  const ref = useRef(null)
   184→  const inView = useInView(ref, { once: true, margin: '-100px' })
   185→  return (
   186→    <motion.div
   187→      ref={ref}
   188→      initial={{ opacity: 0, y: 30 }}
   189→      animate={inView ? { opacity: 1, y: 0 } : {}}
   190→      transition={{ duration: 0.7, delay }}
   191→      className={className}
   192→    >
   193→      {children}
   194→    </motion.div>
   195→  )
   196→}
