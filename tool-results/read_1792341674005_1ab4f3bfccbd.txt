   171→/* --------------------------- helper components ----------------------------- */
   172→
   173→function Reveal({
   174→  children,
   175→  delay = 0,
   176→  className = '',
   177→}: {
   178→  children: React.ReactNode
   179→  delay?: number
   180→  className?: string
   181→}) {
   182→  const ref = useRef(null)
   183→  const inView = useInView(ref, { once: true, margin: '-100px' })
   184→  return (
   185→    <motion.div
   186→      ref={ref}
   187→      initial={{ opacity: 0, y: 30 }}
   188→      animate={inView ? { opacity: 1, y: 0 } : {}}
   189→      transition={{ duration: 0.7, delay }}
   190→      className={className}
   191→    >
   192→      {children}
   193→    </motion.div>
   194→  )
   195→}
