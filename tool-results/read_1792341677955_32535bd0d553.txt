   172→   172→
   173→   173→function Reveal({
   174→   174→  children,
   175→   175→  delay = 0,
   176→   176→  className = '',
   177→   177→}: {
   178→   178→  children: React.ReactNode
   179→   179→  delay?: number
   180→   180→  className?: string
   181→   181→}) {
   182→   182→  const ref = useRef(null)
   183→   183→  const inView = useInView(ref, { once: true, margin: '-100px' })
   184→   184→  return (
   185→   185→    <motion.div
   186→   186→      ref={ref}
   187→   187→      initial={{ opacity: 0, y: 30 }}
   188→   188→      animate={inView ? { opacity: 1, y: 0 } : {}}
   189→   189→      transition={{ duration: 0.7, delay }}
   190→   190→      className={className}
   191→   191→    >
   192→   192→      {children}
   193→   193→    </motion.div>
   194→   194→  )
   195→   195→}
